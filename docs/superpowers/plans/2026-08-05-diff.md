# File-by-file diff Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Diff tab's "File-by-file diff is not implemented yet." placeholder with a per-file list of uncommitted changes whose patches expand inline.

**Architecture:** A new `worktreeDiff(cwd)` in `src/worktree.ts` shells out to git three times (numstat for counts, patch for bodies, `ls-files --others` for untracked) and returns a `FileDiff[]`. A new `/diff?id=` route in `src/server.ts` exposes it, copied from the existing `/graph` route. `web/app.js` renders each file as a native `<details>`; the two pure string helpers it needs live in `web/lib.js` where they can be tested offline.

**Tech Stack:** Node 22.18+ running TypeScript directly (no build step), `node:test` + `node:assert`, `node:child_process.execFile`, plain ES-module browser JS with no framework, plain CSS.

**Spec:** `docs/superpowers/specs/2026-08-05-diff-design.md`

## Global Constraints

- **Node 22.18 or newer.** `.ts` files run directly via `node src/server.ts`; there is no compile step and no bundler.
- **No new dependencies.** `package.json` has exactly two, both agent SDKs.
- **Range is uncommitted only** — `HEAD` to working tree, untracked included. Not `trunk...HEAD`.
- **Pathspec is `-- . ':!uploads'`** (or `':(exclude)uploads'` for `ls-files`), identical to `worktreeStatus()`. `/upload` drops user-attached files there; they are not the agent's work.
- **Never touch the git index.** No `git add -N`. `worktreeStatus()` has an explicit comment promising this; untracked content is read directly instead.
- **Routes are keyed by session id, never a client-supplied path.** The directory to read is whatever `registry.get(id).cwd` already is.
- **Every colour in `web/style.css` outside the token blocks is a `var()` read.** Use `--working`, `--error`, `--muted`, `--well`, `--hairline`, `--mono`. Do not introduce a hex literal.
- **`npm test` must pass before completion** (`node --test test/*.test.ts`). Tests stay offline, use temp directories, and never touch a real account.
- Constants: `MAX_PATCH_LINES = 2000`, patch `maxBuffer = 32 * 1024 * 1024`, untracked file size guard `> 2_000_000` bytes skipped, binary sniff = NUL byte within first 8000 bytes.

```ts
export type FileDiff = {
  path: string;
  old?: string;
  added: number;
  removed: number;
  binary: boolean;
  patch: string;
  truncated: boolean;
};

/** A body longer than this is cut; see the truncation note in the diff spec. */
const MAX_PATCH_LINES = 2000;

/**
 * A patch chunk's own header lines. The pane's summary already names the file,
 * so `index`/`---`/`+++` are noise — but `old mode`, `new file mode`,
 * `rename from/to` and `Binary files … differ` are not, and stay.
 */
const NOISE = /^(index |--- |\+\+\+ )/;

/** Chunk text as it should be rendered: header line dropped, noise filtered. */
function patchBody(chunk: string): string {
  return chunk
    .split('\n')
    .slice(1) // the remainder of the `diff --git a/x b/y` line this was split on
    .filter((l) => !NOISE.test(l))
    .join('\n')
    .replace(/\n+$/, '');
}

function cap(patch: string): { patch: string; truncated: boolean } {
  const lines = patch.split('\n');
  if (lines.length <= MAX_PATCH_LINES) return { patch, truncated: false };
  return { patch: lines.slice(0, MAX_PATCH_LINES).join('\n'), truncated: true };
}

/**
 * `git diff --numstat -z` records. The `-z` form exists because a path may
 * contain a space, a quote or a newline, and the default form escapes those
 * into a quoted string that then has to be unquoted. Its layout is
 * `added \t removed \t path NUL`, except for a rename, which leaves the path
 * field empty and follows with `old NUL new NUL`.
 */
function parseNumstat(stdout: string): Omit<FileDiff, 'patch' | 'truncated'>[] {
  const parts = stdout.split('\0');
  const rows: Omit<FileDiff, 'patch' | 'truncated'>[] = [];
  for (let i = 0; i < parts.length; i++) {
    const [a, r, p] = parts[i].split('\t');
    if (p === undefined) continue; // trailing empty element, not a record
    // Binary files report `-` for both counts; a byte count is not a line count.
    const binary = a === '-';
    const row = { added: binary ? 0 : Number(a), removed: binary ? 0 : Number(r), binary };
    if (p === '') rows.push({ ...row, old: parts[++i], path: parts[++i] });
    else rows.push({ ...row, path: p });
  }
  return rows;
}

/**
 * Every uncommitted change in the worktree, file by file.
 *
 * The range and the pathspec are `worktreeStatus()`'s, deliberately: the pane's
 * per-file numbers and the sidebar's `+N −M` are read inches apart, and a pane
 * that measured a different range would look like a bug rather than a feature.
 */
export async function worktreeDiff(cwd: string): Promise<FileDiff[]> {
  const SPEC = ['--', '.', ':!uploads'];
  const [numstat, patch, others] = await Promise.all([
    // quotePath=false keeps a non-ASCII path readable in the patch headers, which
    // is what the sanity check below matches against.
    run('git', ['-c', 'core.quotePath=false', 'diff', 'HEAD', '--numstat', '-z', ...SPEC], { cwd }),
    // The one call whose output can be large. A generated file blows past
    // execFile's 1MB default maxBuffer and rejects with ENOBUFS, so raise it —
    // and if it still fails, fall back to no bodies rather than losing the list.
    run(
      'git',
      ['-c', 'core.quotePath=false', 'diff', 'HEAD', '--patch', '--no-color', '--no-ext-diff', ...SPEC],
      { cwd, maxBuffer: 32 * 1024 * 1024 },
    ).catch(() => ({ stdout: '' })),
    run('git', ['ls-files', '--others', '--exclude-standard', '-z', '--', '.', ':(exclude)uploads'], { cwd }),
  ]);

  const rows = parseNumstat(numstat.stdout);
  // Zipped by position, not by path: both commands walk the same diff queue with
  // the same pathspec, so they list the same files in the same order — binary
  // files and mode-only changes included. That beats parsing a path back out of
  // a header git may have quoted. The `includes` is the guard on that assumption:
  // if the orders ever diverge, a file shows an empty body rather than somebody
  // else's diff.
  const chunks = patch.stdout.split(/^diff --git /m).slice(1);
  const files: FileDiff[] = rows.map((row, i) => {
    const chunk = chunks[i];
    if (row.binary || !chunk || !chunk.includes(row.path)) {
      return { ...row, patch: '', truncated: false };
    }
    return { ...row, ...cap(patchBody(chunk)) };
  });

  // `git diff HEAD` cannot see untracked files, and leaving them out would show
  // `+0 −0` for the commonest first prompt there is — one telling the agent to
  // *create* files. worktreeStatus() counts their lines directly rather than
  // staging them with `git add -N`; this does the same and builds the body too.
  for (const rel of others.stdout.split('\0')) {
    if (!rel) continue;
    let buf: Buffer;
    try {
      // Size-check first: this runs in the long-lived server process, and an
      // agent that downloaded a dataset into its worktree should not have it
      // read into memory whole.
      const { size } = await fsStat(join(cwd, rel));
      if (size === 0 || size > 2_000_000) continue;
      buf = await readFile(join(cwd, rel));
    } catch {
      continue; // Vanished or unreadable between ls-files and here.
    }
    // git's own binary heuristic: a NUL byte near the start.
    if (buf.subarray(0, 8000).includes(0)) {
      files.push({ path: rel, added: 0, removed: 0, binary: true, patch: '', truncated: false });
      continue;
    }
    const text = buf.toString('utf8');
    const lines = text.split('\n');
    // A file with no trailing newline still ends in a line.
    const n = lines.length - (text.endsWith('\n') ? 1 : 0);
    const body = `@@ -0,0 +1,${n} @@\n${lines.slice(0, n).map((l) => `+${l}`).join('\n')}`;
    files.push({ path: rel, added: n, removed: 0, binary: false, ...cap(body) });
  }

  return files;
}
```

```js
/**
 * The one-line label for a file in the Diff pane. Binary files get a word
 * rather than `+0 −0`: git refuses to count their lines and so does this, and
 * a pair of zeroes would read as "nothing changed".
 */
export function diffSummary(file) {
  const name = file.old ? `${file.old} → ${file.path}` : file.path;
  return file.binary ? `${name} · binary` : `${name} · +${file.added} −${file.removed}`;
}

/**
 * Which class one line of a unified patch body wears. The `---`/`+++` file
 * headers are stripped server-side, so a leading `+` here is always content.
 */
export function patchLineClass(line) {
  if (line.startsWith('@@')) return 'hunk';
  if (line.startsWith('+')) return 'add';
  if (line.startsWith('-')) return 'del';
  return '';
}
```

```js
// Expanded files in the Diff pane, by path. Held here rather than in the DOM
// because the pane is rebuilt whenever the payload changes, which would drop
// the <details> open flags with it. Cleared on session switch by openTab().
const diffOpen = new Set();
// Raw text of the last /diff response, for the identity check in renderDiffPane.
let lastDiffText = '';

function fileBlock(f) {
  const el = document.createElement('details');
  el.className = 'file-diff';
  el.open = diffOpen.has(f.path);
  // <details> rather than a button and a hidden div: the browser already owns
  // the toggle, the keyboard interaction and the aria-expanded state.
  el.ontoggle = () => (el.open ? diffOpen.add(f.path) : diffOpen.delete(f.path));

  const summary = document.createElement('summary');
  summary.textContent = diffSummary(f);

  const body = document.createElement('pre');
  body.className = 'patch';
  if (f.binary) {
    body.textContent = 'Binary file';
  } else {
    const line = (text, cls) => {
      const span = document.createElement('span');
      // textContent, never innerHTML: a patch body is arbitrary repo text and
      // this is the one place it reaches the page.
      span.className = cls;
      span.textContent = `${text}\n`;
      return span;
    };
    const lines = f.patch ? f.patch.split('\n').map((l) => line(l, patchLineClass(l))) : [];
    // Appended as its own element rather than pushed into the text, so a body
    // line that happens to read "… truncated" cannot impersonate the marker.
    if (f.truncated) lines.push(line('… truncated', 'hunk'));
    body.replaceChildren(...lines);
  }

  el.append(summary, body);
  return el;
}

function renderDiffPane(s) {
  const due = forceDiffFetch || Date.now() - lastStatusFetch > STATUS_THROTTLE_MS;
  forceDiffFetch = false;
  if (!due) return;
  lastStatusFetch = Date.now();
  const pane = $('#pane-diff');
  fetch(`/diff?id=${s.id}`).then((r) => r.text()).then((text) => {
    if (openId !== s.id || tab !== 'diff') return;
    // Compared before parsing, and skipped when identical: the diff usually has
    // not changed between two polls, and replacing the DOM anyway would reset
    // the scroll position of a pane the user is reading, every 4 seconds.
    if (text === lastDiffText) return;
    lastDiffText = text;
    const files = JSON.parse(text);
    if (!files.length) {
      const p = document.createElement('p');

}
```
